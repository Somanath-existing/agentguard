Phase 0

What you are building

Read time: 5 min · No code yet

AgentGuard is a thin Python SDK — roughly 600–900 lines of real code — that wraps any LLM agent framework and does three things no existing tool does:

LAYER 1
🔴 Live failure prediction
Scores each tool call between steps. Fires an alert before failure locks in — not after. Research shows 58% of wasted tokens happen after the first detectable warning signal.
LAYER 2
🔁 Semantic crash recovery
Classifies WHY the agent crashed (loop, tool error, hallucination, token overflow) and routes to the right fix — not just "resume from last checkpoint."
LAYER 3
💰 Waste prediction
Estimates token burn for remaining steps before they run. Enforces a budget mid-run. Can downgrade the model automatically when threshold is hit.
THE GAP
🆕 Why this is new
LangSmith/Arize grade after failure. Temporal/LangGraph do raw checkpoints. Langfuse tracks past spend. Nobody unifies live interception + smart recovery + waste prevention in one SDK.

Core data models (models.py)

Define what a "step event" looks like. Every intercepted tool call becomes one of these.

from pydantic import BaseModel
from typing import Optional, Literal
from datetime import datetime
import hashlib

class StepEvent(BaseModel):
    run_id: str
    step_number: int
    span_id: str
    span_name: str                    # e.g. "tool.web_search"
    tool_name: Optional[str] = None
    input_hash: Optional[str] = None  # hash of inputs — used for loop detection
    output_size: int = 0             # chars in output
    tokens_in: int = 0
    tokens_out: int = 0
    latency_ms: float = 0.0
    status: Literal["ok", "error", "timeout"] = "ok"
    error_type: Optional[str] = None
    timestamp: datetime = datetime.utcnow()

    def make_input_hash(self, raw_input: str) -> str:
        return hashlib.sha256(raw_input.encode()).hexdigest()[:16]

class FailureSignal(BaseModel):
    run_id: str
    step_number: int
    signal_type: Literal["loop", "drift", "budget_pressure", "tool_error", "info_collapse"]
    confidence: float          # 0.0–1.0
    detail: str                # human-readable explanation
    recommended_action: Literal["continue", "warn", "redirect", "abort"]
Phase 2 · Weeks 2–3

The Interceptor — hooking into any agent

This is the most important file. Everything else reads from what this captures.

The interceptor is a custom OpenTelemetry SpanProcessor. Every time a tool call starts or ends, OTel calls your hook. This is how you get between steps without touching the agent code.

Key insight

You don't patch LangChain or OpenAI SDK. You register a SpanProcessor once. Because LangChain, OpenAI Agents SDK, and LlamaIndex all emit OTel spans, your interceptor works across all of them with zero framework-specific code.

interceptor.py — the core hook

from opentelemetry.sdk.trace import SpanProcessor
from opentelemetry.sdk.trace import ReadableSpan
from opentelemetry import trace
from .models import StepEvent
from .signals import SignalEngine
from .state import CheckpointStore
from .cost import CostGuardian
import time

class AgentGuardProcessor(SpanProcessor):
    """
    Registers with the OTel TracerProvider.
    on_end() fires after EVERY span in the agent — tool calls,
    LLM calls, retrieval steps. We filter to the ones we care about.
    """

    def __init__(self, session, signal_engine, cost_guardian, store):
        self.session = session
        self.signals = signal_engine
        self.cost = cost_guardian
        self.store = store
        self._step_counts = {}   # run_id → step count

    def on_start(self, span, parent_context=None):
        # Attach our run_id to every span so we can correlate later
        run_id = self.session.run_id
        if run_id:
            span.set_attribute("agentguard.run_id", run_id)

    def on_end(self, span: ReadableSpan):
        # Filter: only process tool calls and LLM calls
        name = span.name
        if not any(k in name for k in ["tool", "llm", "chat", "completion", "retrieve"]):
            return

        run_id = span.attributes.get("agentguard.run_id", "unknown")
        step = self._step_counts.get(run_id, 0) + 1
        self._step_counts[run_id] = step

        # Build a structured StepEvent from the raw OTel span
        event = StepEvent(
            run_id=run_id,
            step_number=step,
            span_id=str(span.context.span_id),
            span_name=name,
            tool_name=span.attributes.get("gen_ai.tool.name"),
            tokens_in=span.attributes.get("gen_ai.usage.input_tokens", 0),
            tokens_out=span.attributes.get("gen_ai.usage.output_tokens", 0),
            output_size=len(str(span.attributes.get("gen_ai.completion", ""))),
            latency_ms=(span.end_time - span.start_time) / 1e6,
            status="error" if span.status.is_error else "ok",
            error_type=str(span.status.description) if span.status.is_error else None
        )

        # Hash the input for loop detection
        raw_input = str(span.attributes.get("gen_ai.prompt", ""))
        event.input_hash = event.make_input_hash(raw_input)

        # Checkpoint state after every step
        self.store.save_step(event)

        # Run signal engine — fire alert if something looks wrong
        signal = self.signals.evaluate(event)
        if signal and signal.recommended_action in ["redirect", "abort"]:
            self.session.on_signal(signal)

        # Check cost budget
        self.cost.record(event)
        if self.cost.is_over_budget():
            self.session.on_budget_exceeded(self.cost.remaining())

    def shutdown(self): pass
    def force_flush(self, timeout_millis=30000): return True

How to register it (in __init__.py)

from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from .interceptor import AgentGuardProcessor
# ... other imports

class Guard:
    def __init__(self, budget_tokens=100000, on_failure="replan", alert_webhook=None):
        self.budget_tokens = budget_tokens
        self.on_failure = on_failure
        self.alert_webhook = alert_webhook

    def watch(self, run_id: str):
        """Context manager. Wraps any agent run."""
        session = GuardSession(run_id, self)
        store = CheckpointStore(run_id)
        cost = CostGuardian(budget_tokens=self.budget_tokens)
        signals = SignalEngine(run_id)
        processor = AgentGuardProcessor(session, signals, cost, store)

        # Register processor with OTel — intercepts ALL spans from now on
        provider = TracerProvider()
        provider.add_span_processor(processor)
        trace.set_tracer_provider(provider)

        # Auto-instrument whichever frameworks are installed
        _auto_instrument()

        return session

def _auto_instrument():
    try:
        from opentelemetry.instrumentation.langchain import LangchainInstrumentor
        LangchainInstrumentor().instrument()
    except ImportError: pass
    try:
        from opentelemetry.instrumentation.openai import OpenAIInstrumentor
        OpenAIInstrumentor().instrument()
    except ImportError: pass
    try:
        from opentelemetry.instrumentation.anthropic import AnthropicInstrumentor
        AnthropicInstrumentor().instrument()
    except ImportError: pass
Phase 3 · Weeks 3–5

Signal Engine — detecting failure mid-run

This is what makes AgentGuard original. Pure math, no LLM call needed.

Three cheap signals that run after every step. None of them require an LLM — they're deterministic math on the step history. Fast enough to run synchronously between tool calls.

🔄
Loop detector
Same input hash called twice = agent is stuck. Fires at 2 repeats.
📉
Info-gain collapse
Output shrinking 3 steps in a row = agent running out of useful things to say.
💸
Budget pressure
Token burn rate vs remaining budget. Projects remaining steps will exceed budget.

signals.py — full implementation

from collections import defaultdict, deque
from .models import StepEvent, FailureSignal
from typing import Optional

class SignalEngine:
    def __init__(self, run_id: str, window: int = 5):
        self.run_id = run_id
        self.window = window
        self.input_hashes: list[str] = []         # for loop detection
        self.output_sizes: deque = deque(maxlen=window) # for info-gain
        self.error_streak: int = 0

    def evaluate(self, event: StepEvent) -> Optional[FailureSignal]:
        # Run all three detectors. Return the highest-severity signal found.
        signals = [
            self._check_loop(event),
            self._check_info_gain(event),
            self._check_error_streak(event),
        ]
        # Filter None, return most severe
        real = [s for s in signals if s is not None]
        if not real:
            return None
        severity_order = {"continue":0, "warn":1, "redirect":2, "abort":3}
        return max(real, key=lambda s: severity_order[s.recommended_action])

    def _check_loop(self, event: StepEvent) -> Optional[FailureSignal]:
        if not event.input_hash:
            return None
        self.input_hashes.append(event.input_hash)

        # Count how many times this exact input appeared in last 10 steps
        recent = self.input_hashes[-10:]
        count = recent.count(event.input_hash)

        if count >= 3:
            return FailureSignal(
                run_id=self.run_id, step_number=event.step_number,
                signal_type="loop", confidence=0.95,
                detail=f"Same input called {count}x in last 10 steps: {event.tool_name}",
                recommended_action="abort"
            )
        if count == 2:
            return FailureSignal(
                run_id=self.run_id, step_number=event.step_number,
                signal_type="loop", confidence=0.65,
                detail=f"Possible loop: same input seen twice for {event.tool_name}",
                recommended_action="warn"
            )
        return None

    def _check_info_gain(self, event: StepEvent) -> Optional[FailureSignal]:
        # Track output size over recent steps
        self.output_sizes.append(event.output_size)
        if len(self.output_sizes) < 4:
            return None

        sizes = list(self.output_sizes)[-4:]
        # Check if output has been monotonically shrinking
        shrinking = all(sizes[i] > sizes[i+1] for i in range(3))
        if shrinking and sizes[-1] < sizes[0] * 0.3:  # 70% output collapse
            return FailureSignal(
                run_id=self.run_id, step_number=event.step_number,
                signal_type="info_collapse", confidence=0.78,
                detail=f"Output shrunk 70% over 4 steps — likely hitting diminishing returns",
                recommended_action="redirect"
            )
        return None

    def _check_error_streak(self, event: StepEvent) -> Optional[FailureSignal]:
        if event.status == "error":
            self.error_streak += 1
        else:
            self.error_streak = 0

        if self.error_streak >= 3:
            return FailureSignal(
                run_id=self.run_id, step_number=event.step_number,
                signal_type="tool_error", confidence=0.90,
                detail=f"3 consecutive errors: {event.error_type}",
                recommended_action="abort"
            )
        if self.error_streak == 2:
            return FailureSignal(
                run_id=self.run_id, step_number=event.step_number,
                signal_type="tool_error", confidence=0.60,
                detail=f"2 consecutive tool errors — escalation likely needed",
                recommended_action="warn"
            )
        return None
💡 Why no LLM call here?

These three signals are deterministic — they run in microseconds between tool calls and have zero additional cost. Save LLM evaluation for the classifier (Phase 4) which only fires on crash or high-confidence abort signal.

Phase 4 · Weeks 5–7

State store + Semantic Recovery Router

This is the second original piece — classifying WHY it failed and picking the right fix

Most frameworks just checkpoint-and-resume. AgentGuard classifies the failure type first, then routes to the appropriate recovery strategy.

Tool error
Retry with backoff
When: API timeout, rate limit, transient error. Resume from last good step.
Loop detected
Replan from checkpoint
When: agent stuck in loop. Restore to last good checkpoint + inject replanning prompt.
Info collapse / hallucination
Escalate to human
When: agent running out of real information. Push to human queue with context.
Token budget exceeded
Downgrade model
When: projected to exceed budget. Swap to cheaper model for remaining steps.

state.py — checkpoint store (SQLite for dev, Redis for prod)

import sqlite3, json
from .models import StepEvent
from typing import Optional

class CheckpointStore:
    """
    Dev: SQLite file per run_id (zero infra required).
    Prod: swap backend to Redis for <1ms reads. Same API.
    """
    def __init__(self, run_id: str, backend: str = "sqlite"):
        self.run_id = run_id
        self.backend = backend
        if backend == "sqlite":
            self.conn = sqlite3.connect(f"agentguard_{run_id}.db")
            self._init_sqlite()

    def _init_sqlite(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS steps (
                step_number INTEGER PRIMARY KEY,
                run_id TEXT,
                data TEXT,          -- JSON-serialized StepEvent
                trust_score REAL DEFAULT 1.0,
                created_at TEXT DEFAULT CURRENT_TIMESTAMP
            )
        """)
        self.conn.commit()

    def save_step(self, event: StepEvent, trust_score: float = 1.0):
        """Save every step. Called from interceptor after each span ends."""
        self.conn.execute(
            "INSERT OR REPLACE INTO steps (step_number, run_id, data, trust_score) VALUES (?,?,?,?)",
            (event.step_number, self.run_id, event.model_dump_json(), trust_score)
        )
        self.conn.commit()

    def last_trusted_step(self, min_trust: float = 0.7) -> Optional[StepEvent]:
        """Find last step with acceptable trust — recovery resumes from here."""
        row = self.conn.execute(
            "SELECT data FROM steps WHERE run_id=? AND trust_score>=? ORDER BY step_number DESC LIMIT 1",
            (self.run_id, min_trust)
        ).fetchone()
        if row:
            return StepEvent.model_validate_json(row[0])
        return None

    def all_steps(self) -> list[StepEvent]:
        rows = self.conn.execute(
            "SELECT data FROM steps WHERE run_id=? ORDER BY step_number",
            (self.run_id,)
        ).fetchall()
        return [StepEvent.model_validate_json(r[0]) for r in rows]

recovery.py — the semantic router

from .models import FailureSignal
from .state import CheckpointStore
import httpx, asyncio

class RecoveryRouter:
    """
    The novel piece: classify failure type, then pick the right fix.
    Not just 'restart from checkpoint' — route to the appropriate strategy.
    """
    def __init__(self, store: CheckpointStore, config: dict):
        self.store = store
        self.config = config
        self._escalation_queue = []

    def handle(self, signal: FailureSignal) -> dict:
        """
        Takes a FailureSignal, returns a recovery action dict.
        The caller (session.py) executes the action.
        """
        last_good = self.store.last_trusted_step()

        if signal.signal_type == "tool_error":
            return {
                "action": "retry",
                "resume_from_step": last_good.step_number if last_good else 0,
                "backoff_ms": 2000,
                "reason": signal.detail
            }

        if signal.signal_type == "loop":
            return {
                "action": "replan",
                "resume_from_step": last_good.step_number if last_good else 0,
                "inject_prompt": "You appear to be repeating the same action. Take a different approach to the goal.",
                "reason": signal.detail
            }

        if signal.signal_type in ["info_collapse", "drift"]:
            self._push_to_human_queue(signal, last_good)
            return {
                "action": "escalate",
                "checkpoint_step": last_good.step_number if last_good else 0,
                "reason": signal.detail
            }

        if signal.signal_type == "budget_pressure":
            return {
                "action": "downgrade_model",
                "new_model": self.config.get("cheap_model", "claude-haiku-4-5-20251001"),
                "reason": signal.detail
            }

        return {"action": "abort", "reason": signal.detail}

    def _push_to_human_queue(self, signal, last_good):
        # In v1: just log it. In v2: POST to webhook or Slack
        self._escalation_queue.append({
            "signal": signal.model_dump(),
            "last_good_step": last_good.step_number if last_good else None,
        })
Phase 5 · Week 7

Cost Guardian — predict waste before it happens

~1 day of code · High impact for users

Existing tools show you what you already spent. The Cost Guardian projects what the remaining steps will cost before they run — and enforces a budget mid-run.

cost.py

from .models import StepEvent
import tiktoken

# Real 2026 pricing per 1M tokens (update from API pricing page)
MODEL_COSTS = {
    "claude-opus-4-6":   {"in": 15.00, "out": 75.00},
    "claude-sonnet-4-6": {"in":  3.00, "out": 15.00},
    "claude-haiku-4-5-20251001":  {"in":  0.80, "out":  4.00},
    "gpt-4o":            {"in":  5.00, "out": 15.00},
    "gpt-4o-mini":       {"in":  0.15, "out":  0.60},
}

class CostGuardian:
    def __init__(self, budget_tokens: int = 100000, budget_usd: float = None):
        self.budget_tokens = budget_tokens
        self.budget_usd = budget_usd
        self.tokens_used: int = 0
        self.usd_used: float = 0.0
        self._history: list[tuple] = []  # (step_num, tokens_in, tokens_out)

    def record(self, event: StepEvent):
        self.tokens_used += event.tokens_in + event.tokens_out
        self._history.append((event.step_number, event.tokens_in, event.tokens_out))

    def is_over_budget(self) -> bool:
        return self.tokens_used >= self.budget_tokens

    def remaining(self) -> int:
        return max(0, self.budget_tokens - self.tokens_used)

    def projected_total(self) -> int:
        """
        Project total token spend at current burn rate.
        If projected > budget * 0.85, return a budget_pressure signal.
        Uses rolling average of last 5 steps.
        """
        if len(self._history) < 3:
            return self.tokens_used

        recent = self._history[-5:]
        avg_per_step = sum(i + o for _, i, o in recent) / len(recent)
        # Assume agent still has ~10 steps left (conservative)
        estimated_remaining_steps = 10
        projected = self.tokens_used + (avg_per_step * estimated_remaining_steps)
        return int(projected)

    def should_warn(self) -> bool:
        """Fire warning at 85% of budget — before it's too late."""
        return self.projected_total() > self.budget_tokens * 0.85

    def cost_usd(self, model: str = "claude-sonnet-4-6") -> float:
        prices = MODEL_COSTS.get(model, MODEL_COSTS["claude-sonnet-4-6"])
        total_in  = sum(i for _, i, _ in self._history) / 1_000_000
        total_out = sum(o for _, _, o in self._history) / 1_000_000
        return (total_in * prices["in"]) + (total_out * prices["out"])

    def summary(self) -> dict:
        return {
            "tokens_used": self.tokens_used,
            "tokens_budget": self.budget_tokens,
            "tokens_remaining": self.remaining(),
            "projected_total": self.projected_total(),
            "over_budget": self.is_over_budget(),
            "estimated_cost_usd": self.cost_usd(),
        }

Terminal dashboard + report

~1 day · Makes your tool feel real to users

Use the rich library to print a live dashboard in the terminal after each run. This is what users screenshot and share.

session.py — the run context + report

from rich.console import Console
from rich.table import Table
from rich.panel import Panel
from rich import print as rprint
from .models import FailureSignal
from .state import CheckpointStore
from .cost import CostGuardian
from contextlib import contextmanager

class GuardSession:
    def __init__(self, run_id: str, guard):
        self.run_id = run_id
        self.guard = guard
        self.signals_fired = []
        self.recovery_actions = []
        self.console = Console()

    def on_signal(self, signal: FailureSignal):
        self.signals_fired.append(signal)
        if signal.recommended_action == "abort":
            self.console.print(f"[bold red]⛔ ABORT signal at step {signal.step_number}: {signal.detail}[/]")
        elif signal.recommended_action == "redirect":
            self.console.print(f"[yellow]⚠️  Redirect signal at step {signal.step_number}: {signal.detail}[/]")

    def on_budget_exceeded(self, remaining: int):
        self.console.print(f"[bold red]💰 Budget exceeded. {remaining} tokens remain in budget.[/]")

    def report(self, cost: CostGuardian = None, store: CheckpointStore = None) -> str:
        # Print a rich table to terminal
        table = Table(title=f"AgentGuard Report — Run {self.run_id}", show_lines=True)
        table.add_column("Metric", style="cyan")
        table.add_column("Value", style="white")

        table.add_row("Signals fired", str(len(self.signals_fired)))
        table.add_row("Recovery actions", str(len(self.recovery_actions)))

        if cost:
            summary = cost.summary()
            table.add_row("Tokens used", f"{summary['tokens_used']:,}")
            table.add_row("Est. cost (USD)", f"${summary['estimated_cost_usd']:.4f}")
            table.add_row("Over budget?", "YES ⛔" if summary['over_budget'] else "No ✅")

        if store:
            steps = store.all_steps()
            table.add_row("Steps recorded", str(len(steps)))

        self.console.print(table)
        return "Report printed"

    def __enter__(self):
        return self

    def __exit__(self, *args):
        self.console.print("[dim]AgentGuard session closed.[/]")
